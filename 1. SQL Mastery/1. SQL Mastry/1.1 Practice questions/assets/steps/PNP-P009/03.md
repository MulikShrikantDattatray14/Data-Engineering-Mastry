```sql
SELECT 'Category' AS source_table,
       'Active' AS flag_column, CAST(Active AS TEXT) AS raw_value , count(*) as row_count
from Category
Group by Active

union all

SELECT 'Payments' AS source_table,
       'Allowed' AS flag_column, CAST(Allowed AS TEXT) AS raw_value , count(*) as row_count
from Payments
group by Allowed
```
